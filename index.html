const form = document.getElementById("chatForm"); const input = document.getElementById("messageInput"); const messages = document.getElementById("messages"); const welcome = document.getElementById("welcome"); const WORKER_URL = "https://ham-ia-cerveau.yv574s6vcn.workers.dev"; function addMessage(text, type) { welcome.style.display = "none"; const bubble = document.createElement("div"); bubble.className = "message " + type; bubble.textContent = text; messages.appendChild(bubble); window.scrollTo(0, document.body.scrollHeight); } form.addEventListener("submit", async function(event) { event.preventDefault(); const text = input.value.trim(); if (!text) return; addMessage(text, "user"); input.value = ""; const loading = document.createElement("div"); loading.className = "message assistant"; loading.textContent = "HAM IA réfléchit..."; messages.appendChild(loading); try { const response = await fetch(WORKER_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: text }) }); const data = await response.json(); loading.remove(); if (!response.ok || !data.reply) { throw new Error(data.error || "Réponse invalide"); } addMessage(data.reply, "assistant"); } catch (error) { loading.remove(); addMessage("Erreur de connexion : " + error.message, "assistant"); } });
